import importlib.util
import json
from pathlib import Path
import pytest

spec = importlib.util.spec_from_file_location('retry', '/private/tmp/judge-retry-first-200.py')
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)

@pytest.mark.parametrize('initial,statuses,expected_calls', [
    (503, [200], 1),
    (503, [503, 200], 2),
    (200, [], 0),
])
def test_first_http200_is_terminal_even_when_validator_fails(tmp_path, monkeypatch,
                                                            initial, statuses, expected_calls):
    route = next(iter(m.ROLES))
    source = tmp_path / 'source'
    folder = source / route
    folder.mkdir(parents=True)
    (folder/'00-request.json').write_text('{}')
    (folder/'result.json').write_text(json.dumps({'planned_packets':1, 'results':[
        {'packet':0, 'http_status':initial, 'contract_pass':False}]}))
    out = tmp_path/'out'
    out.mkdir()
    monkeypatch.setattr(m, 'SOURCE', source)
    monkeypatch.setattr(m, 'OUT', out)
    monkeypatch.setattr(m, 'ROLES', {route:'sibling'})
    monkeypatch.setattr(m.time, 'sleep', lambda _: None)
    calls = []
    def fake(route_id, role, index, number, refresh=False):
        calls.append(number)
        code = statuses[len(calls)-1]
        result={'packet':index,'http_status':code,'http_200_terminal':code==200,
                'retry_in_seconds':0,'evaluation':{'contract_pass':False,'pass':False}}
        directory=out/route_id
        directory.mkdir(exist_ok=True)
        (directory/f'{index:02d}-attempt-{number:03d}.result.json').write_text(json.dumps(result))
        return result
    monkeypatch.setattr(m, 'attempt', fake)
    assert m.main() == 0
    assert len(calls) == expected_calls


def test_shared_account_routes_get_fair_transport_attempts(tmp_path, monkeypatch):
    routes = ['openrouter_google_gemma_4_31b_it_free',
              'openrouter_nvidia_nemotron_3_5_super_vl_free']
    source=tmp_path/'source'
    out=tmp_path/'out'
    out.mkdir()
    for route in routes:
        folder=source/route
        folder.mkdir(parents=True)
        (folder/'00-request.json').write_text('{}')
        (folder/'result.json').write_text(json.dumps({'planned_packets':1,'results':[
            {'packet':0,'http_status':503}]}))
    monkeypatch.setattr(m,'SOURCE',source)
    monkeypatch.setattr(m,'OUT',out)
    monkeypatch.setattr(m,'ROLES',{r:'sibling' for r in routes})
    monkeypatch.setattr(m.time,'sleep',lambda _:None)
    calls=[]
    def fake(route,role,index,number,refresh=False):
        calls.append(route)
        code=503 if route==routes[0] and number==1 else 200
        r={'http_200_terminal':code==200,'http_status':code,'retry_in_seconds':0}
        p=out/route
        p.mkdir(exist_ok=True)
        (p/f'{index:02d}-attempt-{number:03d}.result.json').write_text(json.dumps(r))
        return r
    monkeypatch.setattr(m,'attempt',fake)
    assert m.main()==0
    assert calls==[routes[0],routes[1],routes[0]]
