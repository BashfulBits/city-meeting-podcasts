def wilson(k,n):
 if not n:return None
 z=1.959963984540054;den=1+z*z/n;mid=(k/n+z*z/(2*n))/den;half=z*math.sqrt((k/n*(1-k/n)+z*z/(4*n))/n)/den
 return [max(0,mid-half),min(1,mid+half)]

def score(a,b,group=None):

 return {'n':len(ids),'positives':pos,'negatives':neg,'accepted':len(accepted),'true_accepts':tp,'false_accepts':len(fp),'false_accept_ids':fp,'acceptance_coverage':len(accepted)/len(ids),'positive_recall':tp/pos if pos else None,'false_accept_rate':len(fp)/neg if neg else None,'false_accept_rate_wilson95':wilson(len(fp),neg),'false_discovery_rate':len(fp)/len(accepted) if accepted else None,'false_discovery_wilson95':wilson(len(fp),len(accepted)),'human_review':len(ids)-len(accepted)}
