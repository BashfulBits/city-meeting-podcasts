from pathlib import Path
from datetime import datetime,UTC,timedelta
from citypods.remedy_evaluation import Manifest,RemedyEvalAnswer,case_messages,canonical_hash,validate_answer
from citypods.compute.structured import register_response_model,parse_structured_json
from citypods.compute.base import InferenceJob
from citypods.compute.llm import LiteLLMBackend,LLMBackendConfig
from citypods.compute.llm_policy import LLMRequestPolicy
from citypods.config import load_site_config
from citypods.storage import make_storage
root=Path('/private/tmp/p1-kimi-reviewer')
m=Manifest.model_validate(json.loads((root/'evals/remedy/manifest.json').read_text()));case=m.cases[0]
s=load_site_config(root/'config/site_config.yml');storage=make_storage(s,s.get('base_url',''),root/s.get('output_dir','docs'))
assert storage is not None and storage.cas_capable
register_response_model('remedy-eval-v2',RemedyEvalAnswer)
backend=LiteLLMBackend(LLMBackendConfig(model='nvidia/deepseek-v4.1-flash'),storage=storage)
policy=LLMRequestPolicy(allowed_models=('nvidia/deepseek-v4.1-flash',),allowed_route_ids=('nvidia_deepseek_v4_1_flash_free',),purpose='audit-remedy',require_direct=True,deadline_at=datetime.now(UTC)+timedelta(seconds=150))

 answer=validate_answer(parse_structured_json(result.output['choices'][0]['message']['content']),case)
 row.update(status='completed',route_id=result.route_id,upstream_model=result.upstream_model,reasoning_level=result.reasoning_level,provider_attempts=result.provider_attempts,answer=answer.model_dump(),raw_response=result.output)
